1. CAPTURE: OpenBSM → JSONL SEGMENTS
═══════════════════════════════════════════════════════════════════════════════

FILE: src/bin/capture_macos_rotating.rs (519 lines)

ENTRY POINT:
  fn main()
    ├─ get_telemetry_root() -> PathBuf
    │  └─ Checks CLI --telemetry-root → env EDR_TELEMETRY_ROOT → ./telemetry_output
    ├─ validate_root_permissions()
    │  └─ libc::geteuid() must == 0, else exit(1)
    ├─ fs::create_dir_all(segments_dir)
    ├─ load_or_create_index(index_path) -> SegmentIndex
    │  └─ Parses existing index.json or creates new
    ├─ OpenOptions::open("/dev/auditpipe")
    │  └─ PRODUCTION PATH: Real audit stream
    └─ Loop:
       ├─ auditpipe.read(&mut buffer) -> bytes
       │  └─ [TODO: Parse OpenBSM binary format here]
       ├─ NormalizedEvent {schema_version, ts_ms, host, event_type, subject, process, payload}
       │  └─ Construct normalized event (currently placeholder)
       ├─ current_segment_events.push(event)
       └─ If rotation_ms elapsed or buffer full:
          ├─ write_segment_atomic(segment_dir, segment_id, &events) -> size_bytes
          │  ├─ Open temp file
          │  ├─ For each event: serde_json::to_string + "\n"
          │  ├─ Drop file (flush)
          │  ├─ System "sync" command
          │  └─ fs::rename(temp → final) ATOMIC
          └─ update_index_atomic(index_path, index, segment_id, rel_path, ts_first, ts_last, records, size)
             ├─ Add/update SegmentMetadata in index
             ├─ Write to temp
             ├─ System "sync"
             └─ fs::rename(temp → final) ATOMIC

KEY STRUCTURES:
  SegmentIndex {
    schema_version: u32,
    segments: Vec<SegmentMetadata>,
    last_updated_ts: u64,
  }
  
  SegmentMetadata {
    segment_id, rel_path, ts_first, ts_last, records, size_bytes, sha256,
  }
  
  NormalizedEvent {
    schema_version, ts_ms, host, event_type,
    subject: {uid, gid, auid, euid, egid, session_id},
    process: {pid, ppid, exe, argv, cwd},
    payload: Value (event-type specific),
    raw_bsm: Option<Value>,
  }

ATOMICITY GUARANTEES:
  - Segment writes: JSONL written to temp, synced, renamed (no partial writes)
  - Index updates: JSON written to temp, synced, renamed (no partial reads)
  - Schema versioning: All events include schema_version=1

2. INGESTER: JSONL → SQLite → INCIDENTS
═══════════════════════════════════════════════════════════════════════════════

FILE: src/bin/edr_locald.rs (349 lines)

ENTRY POINT:
  fn main()
    ├─ get_telemetry_root() -> PathBuf [same as capture]
    ├─ Connection::open(analysis.db) + execute_batch(CREATE TABLE IF NOT EXISTS)
    │  ├─ telemetry_events(id, segment_id, record_index, ts, event_type, event_json)
    │  │  └─ Index: UNIQUE(segment_id, record_index) [idempotency]
    │  ├─ locald_checkpoint(segment_id, last_record_index, ts_processed)
    │  │  └─ Restart safety: resume from last processed record
    │  └─ incidents(incident_id PK, playbook_id, primary_entity, ts_created, ts_updated, ...)
    │     └─ Index: (ts_created), (playbook_id), (severity)
    ├─ load_index(telemetry_root) -> SegmentIndex
    └─ Main loop:
       ├─ Watch index.json for new segments
       ├─ For each new segment:
       │  └─ ingest_segment(segment_path, segment_id, db, accumulated_facts)
       │     ├─ Read JSONL file line by line
       │     ├─ For each line:
       │     │  ├─ serde_json::from_str(line) -> Value
       │     │  ├─ Extract event_type from value.event_type
       │     │  ├─ INSERT INTO telemetry_events (segment_id, record_index, ts, event_type, event_json)
       │     │  │  └─ Triggers UNIQUE constraint if duplicate → ignored
       │     │  ├─ extract_facts_from_event(event_type, event_data, ...) -> Vec<Fact>
       │     │  │  ├─ Match on event_type: proc_exec, file_write, net_connect, auth
       │     │  │  └─ For each fact: create Fact {fact_type, exe, user, host, ts}
       │     │  │     + EvidencePtr {segment_id, record_index, ts, event_type}
       │     │  └─ accumulated_facts.push((fact, ptr))
       │     └─ check_playbook_fire(playbook_id, accumulated_facts) -> Option<(evidence_json, slots_json, tags_json)>
       │        ├─ Match facts against playbook rules (demo: pb_demo_proc_write)
       │        ├─ If ALL required slots filled:
       │        │  ├─ Compute incident_id = compute_incident_id(playbook_id, primary_entity, ts_bucket)
       │        │  ├─ INSERT OR REPLACE incidents(incident_id, playbook_id, ..., evidence_json, ...)
       │        │  │  └─ evidence_json = JSON array of EvidencePtrs
       │        │  ├─ UPDATE locald_checkpoint(segment_id, last_record_index, ts_processed)
       │        │  └─ Log: [incident] FIRED
       │        └─ Return None if rules not met

KEY STRUCTURES:
  Fact {
    fact_type: String,  // "ProcExec", "FileWrite", etc.
    exe: String,
    user: String,
    host: String,
    ts: u64,
  }
  
  EvidencePtr {
    segment_id: String,
    record_index: u32,
    ts: u64,
    event_type: String,
  }

DETERMINISTIC INCIDENT ID:
  incident_id = sha256(playbook_id || primary_entity || (ts / 3600) * 3600)
  └─ Bucketed to hour boundary for stability

IDEMPOTENCY:
  - UNIQUE(segment_id, record_index) prevents duplicate events
  - INSERT OR REPLACE on incident_id prevents duplicate incidents
  - locald_checkpoint tracks last processed record per segment

3. API SERVER: DB QUERIES + EVIDENCE DEREF
═══════════════════════════════════════════════════════════════════════════════

FILE: src/bin/ui_server.rs (360 lines)

ENTRY POINT:
  fn main()
    ├─ get_telemetry_root() [same precedence as capture]
    ├─ axum::Router setup
    ├─ GET /api/health → health(State(AppState))
    │  ├─ Check segments dir exists
    │  ├─ Check index.json readable and valid JSON
    │  ├─ Check analysis.db accessible
    │  └─ Return {ok: bool, message: Option<String>}
    ├─ GET /api/incidents → list_incidents(State(AppState))
    │  ├─ Connection::open(analysis.db)
    │  ├─ SELECT incident_id, playbook_id, severity, ts_created, host, user, exe
    │  │         FROM incidents ORDER BY ts_created DESC LIMIT 100
    │  └─ Return {incidents: Vec<Incident>}
    ├─ GET /api/incidents/:id → query_incident(Path(id), State(AppState))
    │  ├─ SELECT * FROM incidents WHERE incident_id = ?
    │  └─ Return full incident row as JSON
    └─ GET /api/incidents/:id/explain → explain_incident(Path(id), State(AppState))
       ├─ SELECT incident_id, playbook_id, evidence_json, slots_json, ... FROM incidents WHERE incident_id = ?
       ├─ Parse evidence_json -> Vec<EvidencePtrJson>
       ├─ For each EvidencePtr:
       │  └─ dereference_evidence(telemetry_root, ptr, conn) -> TimelineEvent
       │     ├─ Try DB first:
       │     │  └─ SELECT event_json FROM telemetry_events
       │     │           WHERE segment_id = ? AND record_index = ?
       │     └─ Fallback to file:
       │        ├─ Open segments/<segment_id>.jsonl
       │        ├─ Seek to line record_index
       │        ├─ Parse line as JSON
       │        └─ Extract exe, user, ts
       ├─ Build timeline:
       │  ├─ Collect all dereferenced TimelineEvents
       │  ├─ Sort by ts
       │  └─ For each event: cite EvidencePtr
       └─ Return ExplanationResponse {
            incident_id,
            playbook_id,
            narrative: String,
            evidence_ptrs: Vec<EvidencePtrJson>,
            timeline: Vec<TimelineEvent>,  // Each item has raw_event JSON
            severity,
          }

EVIDENCE DEREFERENCING (CRITICAL):
  incident_id → evidence_json (array of EvidencePtrs)
    → dereference_evidence(EvidencePtr)
      → query or file lookup (segment_id, record_index)
        → raw_event JSON (with ts, host, exe, user, payload)
          → include in timeline with ptr citation

KEY STRUCTURES:
  ExplanationResponse {
    incident_id: String,
    playbook_id: String,
    narrative: String,
    evidence_ptrs: Vec<EvidencePtrJson>,
    timeline: Vec<TimelineEvent>,
    severity: String,
  }
  
  TimelineEvent {
    ts: u64,
    event_type: String,
    exe: String,
    user: String,
    evidence_ptr: EvidencePtrJson,
    raw_event: Value,  // Full normalized event JSON
  }

4. INCIDENT ID GENERATION (DETERMINISTIC)
═══════════════════════════════════════════════════════════════════════════════

FILE: src/incident_id.rs (50 lines)

FUNCTION:
  fn compute_incident_id(playbook_id: &str, primary_entity: &str, ts: u64) -> String
    ├─ ts_bucket = (ts / 3600) * 3600  // Hour boundary
    ├─ combined = format!("{}||{}||{}", playbook_id, primary_entity, ts_bucket)
    ├─ sha256_hash = Sha256::digest(combined.as_bytes())
    └─ hex_string[0..12] (12-char prefix for readability)

PROPERTIES:
  ✓ Deterministic: same input → same output
  ✓ Stable: incidents within same hour → same ID (upsertable)
  ✓ Unique: different playbooks → different IDs

5. PLAYBOOK MATCHING (SLOT ENGINE)

STRUCTURE (YAML):

  ]
  optional_slots: [

  ]
  tags: [macOS, persistence, T1234]
  narrative_template: String with {slot} references

PLAYBOOKS INCLUDED:

     └─ T1204 (User Execution) + egress

     └─ T1543.001 + T1547.011 (Persistence via LaunchAgent)

  4. sudo_anomaly.yaml
     └─ T1548.003 (Privilege escalation)

FACT MATCHING (in edr_locald):
  check_playbook_fire(playbook_id, accumulated_facts) -> Option<...>
    ├─ For each required_slot:
    │  └─ Find matching Fact in accumulated_facts
    ├─ If all found: playbook fires
    ├─ Compute incident_id (deterministic)

8. DETECTION TEST MATRIX & GAP REPORT
═══════════════════════════════════════════════════════════════════════════════

FILE: scripts/macos/run_detection_matrix.sh (320 lines)

FLOW:
  ├─ Setup clean $TELEMETRY_ROOT
  ├─ Start capture, locald, ui_server
  ├─ For each test case (test_user_writable_exec, test_launch_agent, etc.):
  │  ├─ Execute simulated technique (Atomic Red Team style)
  │  ├─ Wait for ingestion
  │  ├─ Query incident count: SELECT COUNT(*) FROM incidents
  │  ├─ Record: test_id, events observed, incidents fired, root cause
  │  └─ Append to CSV
  ├─ Generate gap_report.csv
  │  └─ Columns: test_id, technique, expected_events, observed_events, 
  │            incident_fired, root_cause, remediation_file
  ├─ Generate gap_report.md (markdown version with summary)
  │  └─ Total tests, passed count, coverage %
  └─ Stop services, output reports

ROOT CAUSE CATEGORIES:
  CAPTURE_GAP: OpenBSM stream doesn't expose event (macOS limitation)
  PARSE_GAP: Capture saw it but normalization failed
  FACT_GAP: Events parsed, but fact extraction incomplete
  PLAYBOOK_GAP: Facts ok, playbook rule missing
  EXPLAIN_GAP: Incident exists, deref/timeline broken
  PASS: Test passed, all stages working

CALL CHAINS: KEY FLOWS
═══════════════════════════════════════════════════════════════════════════════

FLOW 1: Real Event Ingestion
  auditpipe.read() 
    → parse_openBSM_record() [TODO: libbsm bindings]
      → NormalizedEvent {schema_version, ts_ms, host, event_type, ...}
        → write_segment_atomic() [JSONL write + sync + rename]
          → update_index_atomic() [update index.json + sync + rename]
            → index.json updated ✓

FLOW 2: Incident Creation on Fact Match
  load_index()
    → ingest_segment()
      → INSERT telemetry_events [idempotent UNIQUE]
        → extract_facts_from_event()
          → accumulated_facts.push((Fact, EvidencePtr))
            → check_playbook_fire()
              → compute_incident_id() [SHA256 deterministic]
                → INSERT OR REPLACE incidents() [UPSERT]
                  → UPDATE locald_checkpoint() [restart safe]
                    → incident_id in DB ✓

FLOW 3: Evidence Deref (User Query)
  curl /api/incidents/:id/explain
    → explain_incident()
      → SELECT incident FROM DB
        → Parse evidence_json → Vec<EvidencePtr>
          → For each EvidencePtr:
            dereference_evidence()
              → Try: SELECT event_json FROM telemetry_events
              → Fallback: Open segments/:segment_id.jsonl, read line record_index
                → Extract raw_event JSON
                  → TimelineEvent {ts, event_type, evidence_ptr, raw_event}
            → Sort by ts
              → Return timeline with evidence citations ✓
